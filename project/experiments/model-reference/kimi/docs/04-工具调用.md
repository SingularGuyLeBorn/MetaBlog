# 04-工具调用

## 概述

工具调用(Tool Calling / Function Calling)允许 Kimi 模型调用外部工具或函数，扩展模型的能力边界，如查询数据库、调用 API、执行代码等。

## 基础工具调用示例

```python
from openai import OpenAI
import json

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

# 定义工具
tools = [
    {
        "type": "function",
        "function": {
            "name": "getWeather",
            "description": "获取指定城市的天气信息",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {
                        "type": "string",
                        "description": "城市名称，如北京、上海"
                    },
                    "date": {
                        "type": "string",
                        "description": "日期，格式 YYYY-MM-DD"
                    }
                },
                "required": ["city"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "calculate",
            "description": "执行数学计算",
            "parameters": {
                "type": "object",
                "properties": {
                    "expression": {
                        "type": "string",
                        "description": "数学表达式，如 '2 + 2'"
                    }
                },
                "required": ["expression"]
            }
        }
    }
]

# 发送请求
response = client.chat.completions.create(
    model="kimi-k2-turbo-preview",
    messages=[{"role": "user", "content": "北京今天天气怎么样？"}],
    tools=tools,
    tool_choice="auto",  # 让模型自动选择工具
)

# 处理响应
message = response.choices[0].message

if message.tool_calls:
    print("模型请求调用以下工具：")
    for tool_call in message.tool_calls:
        print(f"\n工具名称: {tool_call.function.name}")
        print(f"参数: {tool_call.function.arguments}")
else:
    print(f"模型回复: {message.content}")
```

## 完整工具调用流程

```python
from openai import OpenAI
import json

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

# 模拟工具函数
def getWeather(city: str, date: str = None):
    """模拟天气查询"""
    return {"city": city, "temperature": "25°C", "weather": "晴朗"}

def calculate(expression: str):
    """模拟计算器"""
    try:
        result = eval(expression)
        return {"expression": expression, "result": result}
    except Exception as e:
        return {"error": str(e)}

# 工具映射
tool_functions = {
    "getWeather": getWeather,
    "calculate": calculate,
}

# 定义工具 schema
tools = [
    {
        "type": "function",
        "function": {
            "name": "getWeather",
            "description": "获取指定城市的天气信息",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {"type": "string"},
                    "date": {"type": "string"}
                },
                "required": ["city"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "calculate",
            "description": "执行数学计算",
            "parameters": {
                "type": "object",
                "properties": {
                    "expression": {"type": "string"}
                },
                "required": ["expression"]
            }
        }
    }
]

def chat_with_tools(user_message: str):
    """支持工具调用的对话函数"""
    messages = [
        {"role": "system", "content": "You are Kimi, an AI assistant that can use tools."},
        {"role": "user", "content": user_message}
    ]
    
    # 第一轮：获取工具调用请求
    response = client.chat.completions.create(
        model="kimi-k2-turbo-preview",
        messages=messages,
        tools=tools,
        tool_choice="auto",
    )
    
    assistant_message = response.choices[0].message
    
    # 如果没有工具调用，直接返回结果
    if not assistant_message.tool_calls:
        return assistant_message.content
    
    # 将助手消息添加到历史
    messages.append({
        "role": "assistant",
        "content": assistant_message.content or "",
        "tool_calls": [
            {
                "id": tc.id,
                "type": tc.type,
                "function": {
                    "name": tc.function.name,
                    "arguments": tc.function.arguments
                }
            } for tc in assistant_message.tool_calls
        ]
    })
    
    # 执行工具调用
    for tool_call in assistant_message.tool_calls:
        function_name = tool_call.function.name
        function_args = json.loads(tool_call.function.arguments)
        
        print(f"🔧 执行工具: {function_name}")
        print(f"   参数: {function_args}")
        
        # 调用对应函数
        if function_name in tool_functions:
            result = tool_functions[function_name](**function_args)
        else:
            result = {"error": f"Unknown function: {function_name}"}
        
        print(f"   结果: {result}")
        
        # 将工具结果添加到消息
        messages.append({
            "role": "tool",
            "tool_call_id": tool_call.id,
            "name": function_name,
            "content": json.dumps(result)
        })
    
    # 第二轮：获取最终回复
    final_response = client.chat.completions.create(
        model="kimi-k2-turbo-preview",
        messages=messages,
        tools=tools,
    )
    
    return final_response.choices[0].message.content

# 使用示例
result = chat_with_tools("北京天气怎么样？15乘以23等于多少？")
print(f"\n📄 最终回复: {result}")
```

## 工具调用参数说明

| 参数 | 类型 | 说明 |
|-----|------|------|
| `tools` | array | 工具定义列表，最多 128 个 |
| `tool_choice` | string/object | 工具选择策略：`"auto"`, `"none"`, 或强制指定 |

### tool_choice 选项

- `"auto"`：模型自动决定是否使用工具(默认)
- `"none"`：强制不使用工具
- `{"type": "function", "function": {"name": "function_name"}}`：强制使用指定工具

## 流式工具调用

```python
from openai import OpenAI
import json

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

tools = [{
    "type": "function",
    "function": {
        "name": "get_time",
        "description": "获取当前时间",
        "parameters": {
            "type": "object",
            "properties": {}
        }
    }
}]

response = client.chat.completions.create(
    model="kimi-k2-turbo-preview",
    messages=[{"role": "user", "content": "现在几点了？"}],
    tools=tools,
    stream=True,
)

# 累积工具调用信息
tool_calls_acc = []
current_content = ""

for chunk in response:
    delta = chunk.choices[0].delta
    
    # 累积内容
    if delta.content:
        current_content += delta.content
        print(delta.content, end="", flush=True)
    
    # 累积工具调用
    if delta.tool_calls:
        for tc in delta.tool_calls:
            idx = tc.index
            while len(tool_calls_acc) <= idx:
                tool_calls_acc.append({"id": "", "function": {"name": "", "arguments": ""}})
            
            if tc.id:
                tool_calls_acc[idx]["id"] = tc.id
            if tc.function and tc.function.name:
                tool_calls_acc[idx]["function"]["name"] = tc.function.name
            if tc.function and tc.function.arguments:
                tool_calls_acc[idx]["function"]["arguments"] += tc.function.arguments

# 打印累积的工具调用
if tool_calls_acc:
    print("\n\n累积的工具调用:")
    for tc in tool_calls_acc:
        print(f"  {tc['function']['name']}: {tc['function']['arguments']}")
```

## 工具定义最佳实践

1. **函数名称**：使用有意义的英文单词，遵循 `^[a-zA-Z_][a-zA-Z0-9-_]{0,63}$` 规范
2. **描述清晰**：`description` 应准确描述工具的功能和使用场景
3. **参数明确**：使用 JSON Schema 定义参数类型和约束

```python
# 好的工具定义示例
tools = [{
    "type": "function",
    "function": {
        "name": "search_database",
        "description": "在用户数据库中搜索记录。适用于需要查找特定用户信息、订单历史等场景。",
        "parameters": {
            "type": "object",
            "properties": {
                "query": {
                    "type": "string",
                    "description": "搜索关键词"
                },
                "limit": {
                    "type": "integer",
                    "description": "返回结果数量限制，默认 10",
                    "default": 10,
                    "minimum": 1,
                    "maximum": 100
                }
            },
            "required": ["query"]
        }
    }
}]
```

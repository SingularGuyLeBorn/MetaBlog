# 08-多轮对话

## 概述

多轮对话通过维护对话历史（message history）实现上下文感知，使模型能够理解对话的上下文和背景信息。

## 基础多轮对话

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

# 初始化对话历史
messages = [
    {"role": "system", "content": "You are Kimi, a helpful AI assistant."}
]

def chat(message: str) -> str:
    """发送消息并获取回复"""
    # 添加用户消息
    messages.append({"role": "user", "content": message})
    
    # 发送请求
    response = client.chat.completions.create(
        model="kimi-k2-turbo-preview",
        messages=messages,
    )
    
    # 获取回复
    reply = response.choices[0].message.content
    
    # 添加助手回复到历史
    messages.append({"role": "assistant", "content": reply})
    
    return reply

# 多轮对话示例
print("User: 你好！")
print(f"Kimi: {chat('你好！')}")

print("\nUser: 我叫张三")
print(f"Kimi: {chat('我叫张三')}")

print("\nUser: 我叫什么名字？")
print(f"Kimi: {chat('我叫什么名字？')}")  # 模型能记住之前的对话
```

## 带历史管理的对话类

```python
from openai import OpenAI
from typing import List, Dict, Optional

class KimiChat:
    """Kimi 对话管理类"""
    
    def __init__(
        self,
        api_key: str,
        model: str = "kimi-k2-turbo-preview",
        system_prompt: str = "You are Kimi, a helpful AI assistant.",
        max_history: int = 10
    ):
        self.client = OpenAI(
            api_key=api_key,
            base_url="https://api.moonshot.ai/v1",
        )
        self.model = model
        self.max_history = max_history
        self.messages: List[Dict[str, str]] = []
        
        if system_prompt:
            self.messages.append({"role": "system", "content": system_prompt})
    
    def chat(
        self,
        message: str,
        stream: bool = False,
        **kwargs
    ) -> str:
        """
        发送消息并获取回复
        
        Args:
            message: 用户消息
            stream: 是否流式输出
            **kwargs: 额外的请求参数
        
        Returns:
            助手回复
        """
        # 添加用户消息
        self.messages.append({"role": "user", "content": message})
        
        # 管理历史长度
        self._manage_history()
        
        # 发送请求
        response = self.client.chat.completions.create(
            model=self.model,
            messages=self.messages,
            stream=stream,
            **kwargs
        )
        
        if stream:
            # 流式输出
            reply_parts = []
            for chunk in response:
                content = chunk.choices[0].delta.content
                if content:
                    reply_parts.append(content)
                    print(content, end="", flush=True)
            reply = "".join(reply_parts)
            print()  # 换行
        else:
            reply = response.choices[0].message.content
        
        # 添加助手回复到历史
        self.messages.append({"role": "assistant", "content": reply})
        
        return reply
    
    def _manage_history(self):
        """管理对话历史长度"""
        # 保留 system prompt + 最近 max_history 轮对话
        if len(self.messages) > self.max_history * 2 + 1:
            system = self.messages[0] if self.messages[0]["role"] == "system" else None
            self.messages = self.messages[-(self.max_history * 2):]
            if system:
                self.messages.insert(0, system)
    
    def clear_history(self, keep_system: bool = True):
        """清空对话历史"""
        if keep_system and self.messages and self.messages[0]["role"] == "system":
            system = self.messages[0]
            self.messages = [system]
        else:
            self.messages = []
    
    def get_history(self) -> List[Dict[str, str]]:
        """获取当前对话历史"""
        return self.messages.copy()
    
    def set_system_prompt(self, prompt: str):
        """设置系统提示词"""
        if self.messages and self.messages[0]["role"] == "system":
            self.messages[0]["content"] = prompt
        else:
            self.messages.insert(0, {"role": "system", "content": prompt})

# 使用示例
chat = KimiChat(
    api_key="your-api-key",
    model="kimi-k2-turbo-preview",
    system_prompt="你是一个专业的编程助手。",
    max_history=5
)

# 多轮对话
print("Kimi:", chat.chat("Python 中如何实现单例模式？"))
print("\nKimi:", chat.chat("请给出具体的代码示例"))
print("\nKimi:", chat.chat("这个实现是线程安全的吗？"))
```

## 多轮对话 + 思考模式

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

messages = [
    {"role": "system", "content": "You are a math tutor."}
]

def chat_with_thinking(message: str) -> tuple[str, Optional[str]]:
    """
    带思考模式的对话
    
    Returns:
        (回复内容, 思考过程)
    """
    messages.append({"role": "user", "content": message})
    
    response = client.chat.completions.create(
        model="kimi-k2.5",
        messages=messages,
        thinking={"type": "enabled"},
    )
    
    message_obj = response.choices[0].message
    content = message_obj.content
    reasoning = getattr(message_obj, 'reasoning_content', None)
    
    messages.append({"role": "assistant", "content": content})
    
    return content, reasoning

# 示例
content, reasoning = chat_with_thinking("解方程 x² - 5x + 6 = 0")
if reasoning:
    print(f"🧠 思考: {reasoning[:200]}...")
print(f"📄 回复: {content}")

# 继续对话
content, reasoning = chat_with_thinking("再用求根公式验证一下")
print(f"📄 回复: {content}")
```

## 多轮对话 + 工具调用

```python
from openai import OpenAI
import json

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

tools = [{
    "type": "function",
    "function": {
        "name": "get_weather",
        "description": "获取天气信息",
        "parameters": {
            "type": "object",
            "properties": {
                "city": {"type": "string"}
            },
            "required": ["city"]
        }
    }
}]

messages = [
    {"role": "system", "content": "You are a helpful assistant with weather tool."}
]

def chat_with_tools(user_message: str) -> str:
    """支持工具的多轮对话"""
    messages.append({"role": "user", "content": user_message})
    
    while True:
        response = client.chat.completions.create(
            model="kimi-k2-turbo-preview",
            messages=messages,
            tools=tools,
        )
        
        choice = response.choices[0]
        message = choice.message
        
        # 添加助手消息
        messages.append({
            "role": "assistant",
            "content": message.content or "",
            "tool_calls": [
                {
                    "id": tc.id,
                    "type": tc.type,
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments
                    }
                } for tc in (message.tool_calls or [])
            ] if message.tool_calls else None
        })
        
        # 如果没有工具调用，返回结果
        if not message.tool_calls:
            return message.content
        
        # 执行工具
        for tool_call in message.tool_calls:
            if tool_call.function.name == "get_weather":
                args = json.loads(tool_call.function.arguments)
                # 模拟天气查询
                result = {"city": args["city"], "weather": "晴朗", "temp": "25°C"}
            else:
                result = {"error": "Unknown tool"}
            
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "name": tool_call.function.name,
                "content": json.dumps(result)
            })

# 多轮对话
print("User: 北京天气怎么样？")
print(f"Assistant: {chat_with_tools('北京天气怎么样？')}")

print("\nUser: 上海呢？")
print(f"Assistant: {chat_with_tools('上海呢？')}")  # 能理解"呢"指代天气

print("\nUser: 谢谢")
print(f"Assistant: {chat_with_tools('谢谢')}")
```

## 上下文压缩策略

对于长对话，可以采用以下策略优化：

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

def summarize_history(messages: list) -> list:
    """总结历史对话"""
    if len(messages) <= 4:
        return messages
    
    # 保留 system prompt
    system = messages[0] if messages[0]["role"] == "system" else None
    
    # 需要总结的内容
    to_summarize = messages[1:-2]  # 除首尾外的内容
    
    summary_prompt = f"""
    请总结以下对话的关键信息，保留重要的上下文和事实：
    
    {json.dumps(to_summarize, ensure_ascii=False)}
    
    用一段话总结。
    """
    
    response = client.chat.completions.create(
        model="kimi-k2-turbo-preview",
        messages=[{"role": "user", "content": summary_prompt}],
    )
    
    summary = response.choices[0].message.content
    
    # 重建消息列表
    new_messages = []
    if system:
        new_messages.append(system)
    new_messages.append({"role": "assistant", "content": f"[历史对话总结] {summary}"})
    new_messages.extend(messages[-2:])  # 添加最近的两条消息
    
    return new_messages

# 使用示例
messages = [
    {"role": "system", "content": "You are a helpful assistant."},
    # ... 很多轮对话 ...
]

# 压缩历史
messages = summarize_history(messages)
```

# 02-流式输出

## 概述

流式输出（Streaming）允许在生成过程中逐步接收响应内容，提供更好的用户体验，特别适用于长文本生成场景。

## 基础流式调用

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

response = client.chat.completions.create(
    model="kimi-k2-turbo-preview",
    messages=[
        {"role": "user", "content": "请写一篇关于人工智能的短文"}
    ],
    stream=True,  # 启用流式输出
)

# 逐块接收内容
for chunk in response:
    if chunk.choices[0].delta.content:
        print(chunk.choices[0].delta.content, end="", flush=True)
```

## 完整流式示例

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

def stream_chat(prompt: str):
    """流式对话函数"""
    response = client.chat.completions.create(
        model="kimi-k2-turbo-preview",
        messages=[{"role": "user", "content": prompt}],
        stream=True,
        stream_options={"include_usage": True},  # 包含 token 使用统计
    )
    
    collected_content = []
    
    for chunk in response:
        # 获取内容增量
        delta = chunk.choices[0].delta
        
        if delta.content:
            content = delta.content
            collected_content.append(content)
            print(content, end="", flush=True)
        
        # 最后一块包含 usage 信息
        if chunk.usage:
            print(f"\n\nToken 使用统计:")
            print(f"  Prompt tokens: {chunk.usage.prompt_tokens}")
            print(f"  Completion tokens: {chunk.usage.completion_tokens}")
            print(f"  Total tokens: {chunk.usage.total_tokens}")
    
    return "".join(collected_content)

# 使用示例
result = stream_chat("解释什么是机器学习")
```

## 流式响应格式

流式响应中的每个 chunk 结构：

```python
{
    "id": "chatcmpl-xxx",
    "object": "chat.completion.chunk",
    "created": 1699999999,
    "model": "kimi-k2-turbo-preview",
    "choices": [
        {
            "index": 0,
            "delta": {
                "role": "assistant",  # 仅在第一个 chunk
                "content": "生成的文本片段"
            },
            "finish_reason": None  # 最后一个 chunk 为 "stop"
        }
    ],
    "usage": None  # 最后一个 chunk 包含统计
}
```

## 带打字机效果的流式输出

```python
import time
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

def typewriter_effect(prompt: str, delay: float = 0.02):
    """带打字机效果的流式输出"""
    response = client.chat.completions.create(
        model="kimi-k2-turbo-preview",
        messages=[{"role": "user", "content": prompt}],
        stream=True,
    )
    
    for chunk in response:
        content = chunk.choices[0].delta.content
        if content:
            print(content, end="", flush=True)
            time.sleep(delay)  # 模拟打字机效果

# 使用示例
typewriter_effect("讲一个短故事", delay=0.03)
```

## 流式输出与思考模式

Kimi K2.5 的思考模式也支持流式输出：

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[{"role": "user", "content": "解方程 2x + 5 = 13"}],
    thinking={"type": "enabled"},  # 启用思考模式
    stream=True,
)

print("思考过程：")
print("-" * 50)

for chunk in response:
    # 思考内容的流式输出
    if hasattr(chunk.choices[0].delta, 'reasoning_content'):
        reasoning = chunk.choices[0].delta.reasoning_content
        if reasoning:
            print(reasoning, end="", flush=True)
    
    # 正式回答的流式输出
    content = chunk.choices[0].delta.content
    if content:
        print(content, end="", flush=True)
```

## 流式输出与工具调用

注意：工具调用模式下，流式输出的处理与非流式类似，但需要正确处理 `tool_calls` 字段的增量更新。

```python
from openai import OpenAI
import json

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.ai/v1",
)

response = client.chat.completions.create(
    model="kimi-k2-turbo-preview",
    messages=[{"role": "user", "content": "今天北京天气如何？"}],
    tools=[{
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "获取天气信息",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {"type": "string"}
                }
            }
        }
    }],
    stream=True,
)

# 累积工具调用信息
tool_calls_accumulator = []

for chunk in response:
    delta = chunk.choices[0].delta
    
    # 累积工具调用
    if delta.tool_calls:
        for tc in delta.tool_calls:
            if len(tool_calls_accumulator) <= tc.index:
                tool_calls_accumulator.append({"id": "", "function": {"name": "", "arguments": ""}})
            if tc.id:
                tool_calls_accumulator[tc.index]["id"] = tc.id
            if tc.function.name:
                tool_calls_accumulator[tc.index]["function"]["name"] = tc.function.name
            if tc.function.arguments:
                tool_calls_accumulator[tc.index]["function"]["arguments"] += tc.function.arguments
    
    # 打印普通内容
    if delta.content:
        print(delta.content, end="", flush=True)

# 处理累积的工具调用
if tool_calls_accumulator:
    print("\n\n工具调用请求：")
    for tc in tool_calls_accumulator:
        print(f"  函数: {tc['function']['name']}")
        print(f"  参数: {tc['function']['arguments']}")
```

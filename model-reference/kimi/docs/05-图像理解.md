# 05-图像理解

## 概述

Kimi K2.5 和 vision-preview 系列模型支持图像理解能力，可以分析图片内容并回答相关问题。

## 支持的模型

| 模型名称 | 说明 |
|---------|------|
| `kimi-k2.5` | 多模态模型，原生支持图像理解 |
| `moonshot-v1-8k-vision-preview` | 视觉版 8K 上下文 |
| `moonshot-v1-32k-vision-preview` | 视觉版 32K 上下文 |
| `moonshot-v1-128k-vision-preview` | 视觉版 128K 上下文 |

## 支持的图像格式

- PNG (`image/png`)
- JPEG (`image/jpeg`)
- WebP (`image/webp`)
- GIF (`image/gif`)

**推荐分辨率**：不超过 4K (4096×2160)

## 基础图像理解示例

```python
import os
import base64
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

# 读取并编码图片
def encode_image(image_path: str) -> str:
    """将图片编码为 base64"""
    with open(image_path, "rb") as f:
        return base64.b64encode(f.read()).decode("utf-8")

image_path = "example.jpg"
image_base64 = encode_image(image_path)
extension = os.path.splitext(image_path)[1].replace(".", "")

# 构建图像 URL
image_url = f"data:image/{extension};base64,{image_base64}"

# 发送请求
response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[
        {"role": "system", "content": "You are Kimi, an AI assistant that can understand images."},
        {
            "role": "user",
            "content": [
                {"type": "image_url", "image_url": {"url": image_url}},
                {"type": "text", "text": "请描述这张图片的内容。"}
            ]
        }
    ],
)

print(response.choices[0].message.content)
```

## 多图输入

```python
import os
import base64
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

def encode_image(image_path: str) -> str:
    with open(image_path, "rb") as f:
        ext = os.path.splitext(image_path)[1].replace(".", "")
        base64_data = base64.b64encode(f.read()).decode("utf-8")
        return f"data:image/{ext};base64,{base64_data}"

# 多张图片
image_paths = ["image1.jpg", "image2.png", "image3.webp"]
image_contents = [
    {"type": "image_url", "image_url": {"url": encode_image(path)}}
    for path in image_paths
]

# 添加文本问题
content = image_contents + [{"type": "text", "text": "比较这三张图片的异同。"}]

response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[
        {"role": "user", "content": content}
    ],
)

print(response.choices[0].message.content)
```

## 图像理解 + 流式输出

```python
import os
import base64
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

# 编码图片
with open("example.jpg", "rb") as f:
    image_base64 = base64.b64encode(f.read()).decode("utf-8")

response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[
        {
            "role": "user",
            "content": [
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{image_base64}"}},
                {"type": "text", "text": "详细描述这张图片。"}
            ]
        }
    ],
    stream=True,
)

# 流式接收响应
for chunk in response:
    content = chunk.choices[0].delta.content
    if content:
        print(content, end="", flush=True)
```

## 图像理解 + 思考模式

Kimi K2.5 支持在图像理解时使用思考模式：

```python
import os
import base64
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

with open("math_problem.jpg", "rb") as f:
    image_base64 = base64.b64encode(f.read()).decode("utf-8")

response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[
        {
            "role": "user",
            "content": [
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{image_base64}"}},
                {"type": "text", "text": "解答图片中的数学题，并展示推理过程。"}
            ]
        }
    ],
    thinking={"type": "enabled"},  # 启用思考模式
)

message = response.choices[0].message

# 获取思考过程
if hasattr(message, 'reasoning_content'):
    print("🧠 思考过程：")
    print(message.reasoning_content)
    print()

print("📄 答案：")
print(message.content)
```

## 图像理解 + 工具调用

```python
import os
import base64
from openai import OpenAI
import json

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

with open("chart.jpg", "rb") as f:
    image_base64 = base64.b64encode(f.read()).decode("utf-8")

tools = [{
    "type": "function",
    "function": {
        "name": "analyze_data",
        "description": "分析图表数据",
        "parameters": {
            "type": "object",
            "properties": {
                "chart_type": {"type": "string"},
                "data_summary": {"type": "string"}
            },
            "required": ["chart_type"]
        }
    }
}]

response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[
        {
            "role": "user",
            "content": [
                {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{image_base64}"}},
                {"type": "text", "text": "分析这张图表，并使用工具记录分析结果。"}
            ]
        }
    ],
    tools=tools,
)

message = response.choices[0].message

if message.tool_calls:
    print("🔧 工具调用请求：")
    for tc in message.tool_calls:
        print(f"  {tc.function.name}: {tc.function.arguments}")
else:
    print("📄 分析结果：")
    print(message.content)
```

## 使用文件 ID 引用图片

对于大图片或需要多次引用的图片，建议先上传文件：

```python
from openai import OpenAI

client = OpenAI(
    api_key="your-api-key",
    base_url="https://api.moonshot.cn/v1",
)

# 上传图片文件
with open("large_image.jpg", "rb") as f:
    file = client.files.create(file=f, purpose="file-extract")

file_id = file.id
print(f"文件 ID: {file_id}")

# 使用文件 ID 进行图像理解
response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[
        {
            "role": "user",
            "content": [
                {"type": "image_url", "image_url": {"url": f"ms://{file_id}"}},
                {"type": "text", "text": "描述这张图片。"}
            ]
        }
    ],
)

print(response.choices[0].message.content)
```

## Token 计费说明

图像理解的 token 消耗根据图像分辨率动态计算：

- 分辨率越高，消耗的 token 越多
- 可以使用 Token 估算 API 预先计算
- 计费方式与标准模型相同，按总 token 数计费

```python
# Token 估算示例
response = client.chat.completions.create(
    model="kimi-k2.5",
    messages=[messages],
    max_completion_tokens=1,  # 只估算输入
)

print(f"图像理解预计消耗: {response.usage.prompt_tokens} tokens")
```

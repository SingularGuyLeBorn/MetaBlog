## GRPO(Group Relative Policy Optimization)算法详解

GRPO是一种基于群体的相对策略优化算法，它结合了策略梯度方法和群体智能思想，特别适合多智能体环境和复杂决策问题。

### GRPO的核心思想
1. **群体学习**：多个智能体同时学习，相互借鉴经验
2. **相对评估**：基于群体表现评估个体策略
3. **协同进化**：智能体在竞争中合作，共同进化

以下是GRPO算法的完整Python实现，包含详细注释：

```python
"""
GRPO (Group Relative Policy Optimization) 算法实现
作者：AI助手
日期：2024-12-20

算法特点：
1. 基于群体的策略优化
2. 相对性能评估机制
3. 自适应学习率调整
4. 支持连续和离散动作空间
"""

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from torch.distributions import Categorical, Normal
from collections import deque
import random
from typing import List, Tuple, Dict, Any

class ActorNetwork(nn.Module):
    """演员网络：输出动作策略"""
    def __init__(self, state_dim: int, action_dim: int, hidden_dim: int = 256):
        super().__init__()
        self.network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim)
        )
        
        # 对于连续动作空间，还需要输出标准差
        self.log_std = nn.Parameter(torch.zeros(1, action_dim))
    
    def forward(self, state: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """前向传播，返回动作均值和标准差"""
        mean = self.network(state)
        std = torch.exp(self.log_std).expand_as(mean)
        return mean, std

class CriticNetwork(nn.Module):
    """评论家网络：评估状态价值"""
    def __init__(self, state_dim: int, hidden_dim: int = 256):
        super().__init__()
        self.network = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, 1)
        )
    
    def forward(self, state: torch.Tensor) -> torch.Tensor:
        """前向传播，返回状态价值"""
        return self.network(state)

class GRPOAgent:
    """GRPO智能体"""
    def __init__(
        self,
        state_dim: int,
        action_dim: int,
        is_continuous: bool = True,
        learning_rate: float = 3e-4,
        gamma: float = 0.99,
        gae_lambda: float = 0.95,
        clip_ratio: float = 0.2,
        entropy_coef: float = 0.01,
        group_size: int = 10
    ):
        """
        初始化GRPO智能体
        
        参数:
            state_dim: 状态维度
            action_dim: 动作维度
            is_continuous: 是否为连续动作空间
            learning_rate: 学习率
            gamma: 折扣因子
            gae_lambda: GAE lambda参数
            clip_ratio: PPO裁剪比例
            entropy_coef: 熵系数(鼓励探索)
            group_size: 群体大小
        """
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.is_continuous = is_continuous
        self.gamma = gamma
        self.gae_lambda = gae_lambda
        self.clip_ratio = clip_ratio
        self.entropy_coef = entropy_coef
        self.group_size = group_size
        
        # 初始化网络
        self.actor = ActorNetwork(state_dim, action_dim)
        self.critic = CriticNetwork(state_dim)
        
        # 优化器
        self.actor_optimizer = optim.Adam(self.actor.parameters(), lr=learning_rate)
        self.critic_optimizer = optim.Adam(self.critic.parameters(), lr=learning_rate)
        
        # 经验回放缓冲区
        self.buffer = deque(maxlen=10000)
        
        # 群体存储(用于相对评估)
        self.group_buffer = []
        
    def select_action(self, state: np.ndarray) -> Tuple[np.ndarray, float, torch.Tensor]:
        """
        根据当前状态选择动作
        
        参数:
            state: 当前状态
            
        返回:
            action: 选择的动作
            log_prob: 动作的对数概率
            value: 状态价值估计
        """
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        
        with torch.no_grad():
            # 获取动作分布
            if self.is_continuous:
                mean, std = self.actor(state_tensor)
                dist = Normal(mean, std)
                action = dist.sample()
                log_prob = dist.log_prob(action).sum(-1)
            else:
                logits = self.actor(state_tensor)
                dist = Categorical(logits=logits)
                action = dist.sample()
                log_prob = dist.log_prob(action)
            
            # 获取状态价值
            value = self.critic(state_tensor)
        
        return action.numpy().flatten(), log_prob.item(), value.item()
    
    def compute_gae(
        self,
        rewards: List[float],
        values: List[float],
        dones: List[bool],
        next_value: float
    ) -> Tuple[List[float], List[float]]:
        """
        计算广义优势估计(GAE)
        
        参数:
            rewards: 奖励序列
            values: 价值估计序列
            dones: 终止标志序列
            next_value: 下一个状态的价值
            
        返回:
            advantages: 优势函数估计
            returns: 回报估计
        """
        advantages = []
        gae = 0
        returns = []
        
        # 反向计算GAE
        for t in reversed(range(len(rewards))):
            if t == len(rewards) - 1:
                next_value_t = next_value
                next_non_terminal = 1.0 - dones[t]
            else:
                next_value_t = values[t + 1]
                next_non_terminal = 1.0 - dones[t]
            
            # TD误差
            delta = rewards[t] + self.gamma * next_value_t * next_non_terminal - values[t]
            
            # GAE
            gae = delta + self.gamma * self.gae_lambda * next_non_terminal * gae
            advantages.insert(0, gae)
            
            # 回报
            returns.insert(0, gae + values[t])
        
        return advantages, returns
    
    def compute_relative_advantages(
        self,
        advantages: List[float],
        group_advantages: List[List[float]]
    ) -> List[float]:
        """
        计算相对优势(群体相对评估)
        
        参数:
            advantages: 当前智能体的优势序列
            group_advantages: 群体中所有智能体的优势序列
            
        返回:
            relative_advantages: 相对优势序列
        """
        if not group_advantages:
            return advantages
        
        # 计算群体平均优势
        group_mean_advantages = []
        for i in range(len(advantages)):
            group_adv_at_step = [adv[i] for adv in group_advantages if i < len(adv)]
            if group_adv_at_step:
                group_mean_advantages.append(np.mean(group_adv_at_step))
            else:
                group_mean_advantages.append(0.0)
        
        # 计算相对优势
        relative_advantages = []
        for i in range(len(advantages)):
            if group_mean_advantages[i] != 0:
                # 相对于群体平均表现的优势
                rel_adv = advantages[i] / (group_mean_advantages[i] + 1e-8)
            else:
                rel_adv = advantages[i]
            relative_advantages.append(rel_adv)
        
        return relative_advantages
    
    def update(
        self,
        states: List[np.ndarray],
        actions: List[np.ndarray],
        old_log_probs: List[float],
        advantages: List[float],
        returns: List[float],
        group_data: List[Dict[str, Any]] = None
    ):
        """
        更新策略网络和价值网络
        
        参数:
            states: 状态序列
            actions: 动作序列
            old_log_probs: 旧策略的对数概率
            advantages: 优势函数估计
            returns: 回报估计
            group_data: 群体数据(用于相对评估)
        """
        # 转换为张量
        states_tensor = torch.FloatTensor(np.array(states))
        actions_tensor = torch.FloatTensor(np.array(actions))
        old_log_probs_tensor = torch.FloatTensor(old_log_probs)
        advantages_tensor = torch.FloatTensor(advantages)
        returns_tensor = torch.FloatTensor(returns)
        
        # 如果提供了群体数据，计算相对优势
        if group_data is not None:
            group_advantages = [data['advantages'] for data in group_data]
            relative_advantages = self.compute_relative_advantages(advantages, group_advantages)
            advantages_tensor = torch.FloatTensor(relative_advantages)
        
        # 归一化优势
        advantages_tensor = (advantages_tensor - advantages_tensor.mean()) / (advantages_tensor.std() + 1e-8)
        
        # 多次更新(PPO风格)
        for _ in range(10):  # PPO更新次数
            # 获取当前策略的动作分布
            if self.is_continuous:
                mean, std = self.actor(states_tensor)
                dist = Normal(mean, std)
                new_log_probs = dist.log_prob(actions_tensor).sum(dim=-1)
            else:
                logits = self.actor(states_tensor)
                dist = Categorical(logits=logits)
                new_log_probs = dist.log_prob(actions_tensor)
            
            # 计算概率比
            ratio = torch.exp(new_log_probs - old_log_probs_tensor)
            
            # 计算裁剪的PPO目标
            surr1 = ratio * advantages_tensor
            surr2 = torch.clamp(ratio, 1 - self.clip_ratio, 1 + self.clip_ratio) * advantages_tensor
            actor_loss = -torch.min(surr1, surr2).mean()
            
            # 添加熵正则化
            entropy = dist.entropy().mean()
            actor_loss -= self.entropy_coef * entropy
            
            # 更新演员网络
            self.actor_optimizer.zero_grad()
            actor_loss.backward()
            torch.nn.utils.clip_grad_norm_(self.actor.parameters(), 0.5)
            self.actor_optimizer.step()
            
            # 更新评论家网络
            values = self.critic(states_tensor).squeeze()
            critic_loss = nn.MSELoss()(values, returns_tensor)
            
            self.critic_optimizer.zero_grad()
            critic_loss.backward()
            torch.nn.utils.clip_grad_norm_(self.critic.parameters(), 0.5)
            self.critic_optimizer.step()
    
    def train_episode(
        self,
        env,
        max_steps: int = 1000,
        collect_group_data: bool = False
    ) -> Dict[str, Any]:
        """
        训练一个回合
        
        参数:
            env: 环境
            max_steps: 最大步数
            collect_group_data: 是否收集群体数据
            
        返回:
            episode_data: 回合数据
        """
        state = env.reset()
        episode_reward = 0
        done = False
        steps = 0
        
        # 存储轨迹数据
        states, actions, rewards, dones, values, log_probs = [], [], [], [], [], []
        
        while not done and steps < max_steps:
            # 选择动作
            action, log_prob, value = self.select_action(state)
            
            # 执行动作
            next_state, reward, done, _ = env.step(action)
            
            # 存储数据
            states.append(state)
            actions.append(action)
            rewards.append(reward)
            dones.append(done)
            values.append(value)
            log_probs.append(log_prob)
            
            # 更新状态
            state = next_state
            episode_reward += reward
            steps += 1
        
        # 计算最后一个状态的价值
        if done:
            next_value = 0
        else:
            next_state_tensor = torch.FloatTensor(state).unsqueeze(0)
            with torch.no_grad():
                next_value = self.critic(next_state_tensor).item()
        
        # 计算GAE和回报
        advantages, returns = self.compute_gae(rewards, values, dones, next_value)
        
        # 存储到缓冲区
        for i in range(len(states)):
            self.buffer.append({
                'state': states[i],
                'action': actions[i],
                'reward': rewards[i],
                'next_state': next_state if i == len(states) - 1 else states[i + 1],
                'done': dones[i],
                'log_prob': log_probs[i],
                'value': values[i],
                'advantage': advantages[i],
                'return': returns[i]
            })
        
        episode_data = {
            'states': states,
            'actions': actions,
            'rewards': rewards,
            'dones': dones,
            'values': values,
            'log_probs': log_probs,
            'advantages': advantages,
            'returns': returns,
            'total_reward': episode_reward,
            'steps': steps
        }
        
        # 如果收集群体数据，存储到群体缓冲区
        if collect_group_data:
            self.group_buffer.append(episode_data)
            if len(self.group_buffer) > self.group_size:
                self.group_buffer.pop(0)
        
        return episode_data
    
    def train_with_group(
        self,
        env,
        num_episodes: int = 1000,
        update_frequency: int = 10
    ):
        """
        使用群体数据进行训练
        
        参数:
            env: 环境
            num_episodes: 训练回合数
            update_frequency: 更新频率(每N个回合更新一次)
        """
        episode_rewards = []
        
        for episode in range(num_episodes):
            # 收集数据
            collect_group_data = (episode % update_frequency == 0)
            episode_data = self.train_episode(env, collect_group_data=collect_group_data)
            
            episode_rewards.append(episode_data['total_reward'])
            
            # 定期更新
            if len(self.buffer) >= 1000 and episode % update_frequency == 0:
                # 从缓冲区采样
                batch_size = min(256, len(self.buffer))
                batch = random.sample(self.buffer, batch_size)
                
                # 提取数据
                states = [item['state'] for item in batch]
                actions = [item['action'] for item in batch]
                old_log_probs = [item['log_prob'] for item in batch]
                advantages = [item['advantage'] for item in batch]
                returns = [item['return'] for item in batch]
                
                # 使用群体数据进行更新
                group_data = self.group_buffer if self.group_buffer else None
                self.update(states, actions, old_log_probs, advantages, returns, group_data)
                
                # 清空群体缓冲区(可选)
                if episode % (update_frequency * 10) == 0:
                    self.group_buffer.clear()
            
            # 打印进度
            if episode % 100 == 0:
                avg_reward = np.mean(episode_rewards[-100:]) if len(episode_rewards) >= 100 else np.mean(episode_rewards)
                print(f"Episode {episode}, Reward: {episode_data['total_reward']:.2f}, Avg Reward: {avg_reward:.2f}")
        
        return episode_rewards

class GRPOTrainer:
    """GRPO训练器：管理多个智能体的训练"""
    def __init__(
        self,
        num_agents: int,
        state_dim: int,
        action_dim: int,
        env_factory,
        **agent_kwargs
    ):
        """
        初始化GRPO训练器
        
        参数:
            num_agents: 智能体数量
            state_dim: 状态维度
            action_dim: 动作维度
            env_factory: 环境工厂函数
            **agent_kwargs: 传递给GRPOAgent的参数
        """
        self.num_agents = num_agents
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.env_factory = env_factory
        
        # 创建多个智能体
        self.agents = [
            GRPOAgent(state_dim, action_dim, **agent_kwargs)
            for _ in range(num_agents)
        ]
        
        # 创建多个环境
        self.envs = [env_factory() for _ in range(num_agents)]
        
        # 训练历史
        self.training_history = []
    
    def train(self, num_episodes: int = 1000):
        """
        训练所有智能体
        
        参数:
            num_episodes: 每个智能体的训练回合数
        """
        for episode in range(num_episodes):
            episode_results = []
            
            # 每个智能体收集一个回合的数据
            for i, (agent, env) in enumerate(zip(self.agents, self.envs)):
                episode_data = agent.train_episode(env, collect_group_data=True)
                episode_results.append(episode_data)
            
            # 交换群体数据
            for i, agent in enumerate(self.agents):
                # 收集其他智能体的数据
                group_data = []
                for j, other_agent in enumerate(self.agents):
                    if i != j and hasattr(other_agent, 'group_buffer') and other_agent.group_buffer:
                        # 取最近的数据
                        if other_agent.group_buffer:
                            group_data.append(other_agent.group_buffer[-1])
                
                # 如果有足够的数据，进行更新
                if len(agent.buffer) >= 1000:
                    batch_size = min(256, len(agent.buffer))
                    batch = random.sample(agent.buffer, batch_size)
                    
                    states = [item['state'] for item in batch]
                    actions = [item['action'] for item in batch]
                    old_log_probs = [item['log_prob'] for item in batch]
                    advantages = [item['advantage'] for item in batch]
                    returns = [item['return'] for item in batch]
                    
                    # 使用群体数据进行更新
                    agent.update(states, actions, old_log_probs, advantages, returns, group_data)
            
            # 记录训练进度
            avg_reward = np.mean([data['total_reward'] for data in episode_results])
            self.training_history.append({
                'episode': episode,
                'avg_reward': avg_reward,
                'agent_rewards': [data['total_reward'] for data in episode_results]
            })
            
            # 打印进度
            if episode % 100 == 0:
                print(f"Episode {episode}, Average Reward: {avg_reward:.2f}")
        
        return self.training_history

    class SimpleContinuousEnv:
        """简单的连续控制环境示例"""

            self.state_dim = state_dim

            self.step_count = 0
            self.max_steps = 200
        
        def reset(self):
            self.state = np.random.randn(self.state_dim)
            self.step_count = 0
            return self.state.copy()

            # 简单的动力学：状态 = 状态 + 动作 + 噪声
            noise = np.random.randn(self.state_dim) * 0.1
            self.state = self.state + action + noise
            
            # 简单的奖励：鼓励状态接近原点
            reward = -np.sum(self.state ** 2)
            
            self.step_count += 1
            done = self.step_count >= self.max_steps
            
            return self.state.copy(), reward, done, {}
        
        @property
        def observation_space(self):
            return self.state_dim
        
        @property
        def action_space(self):
            return self.action_dim
    
    # 创建环境工厂
```

## 代码说明

### 主要组件

1. **ActorNetwork**：策略网络，输出动作分布
2. **CriticNetwork**：价值网络，评估状态价值
3. **GRPOAgent**：单个GRPO智能体实现
4. **GRPOTrainer**：多智能体训练管理器

### 关键特性

1. **群体相对评估**：智能体基于群体表现进行相对评估
2. **GAE计算**：使用广义优势估计减少方差
3. **PPO裁剪**：确保策略更新稳定
4. **熵正则化**：鼓励探索，防止过早收敛
5. **经验回放**：提高样本效率

### 算法优势

1. **更好的探索**：群体智能促进多样性探索
2. **稳定训练**：相对评估减少方差
